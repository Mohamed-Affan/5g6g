% EXPERIMENT 1
% Simulation and Comparative Analysis of FDMA, TDMA and CDMA
% Resource Allocation, User Capacity and Mobile-Generation Context

clear;
clc;
close all;

%% PROGRAM 1: CAPACITY ESTIMATION OF FDMA, TDMA AND CDMA

% System Parameters
totalBandwidthHz = 1e6;       % Total bandwidth: 1 MHz
fdmaChannelHz = 200e3;        % FDMA channel width: 200 kHz
tdmaSlots = 8;                % Number of TDMA slots per frame
processingGain = 64;          % CDMA processing gain
targetSIRdB = 7;              % Target CDMA SIR in dB

% Calculate User Capacity
fdmaUsers = floor(totalBandwidthHz / fdmaChannelHz);
tdmaUsers = tdmaSlots;

targetSIR = 10^(targetSIRdB / 10);
cdmaUsers = floor(1 + processingGain / targetSIR);

% Display Results
fprintf('\n============================================\n');
fprintf(' FDMA, TDMA AND CDMA CAPACITY COMPARISON\n');
fprintf('============================================\n');

fprintf('FDMA: %d users\n', fdmaUsers);
fprintf('TDMA: %d users\n', tdmaUsers);
fprintf('CDMA: %d users at a target SIR of %.1f dB\n', ...
    cdmaUsers, targetSIRdB);

% Calculate CDMA SIR for Different User Counts
K = 2:30;
sirLinear = processingGain ./ (K - 1);
sirDb = 10 * log10(sirLinear);

% Display Comparison Table
method = {'FDMA'; 'TDMA'; 'CDMA'};
supportedUsers = [fdmaUsers; tdmaUsers; cdmaUsers];

results = table(method, supportedUsers);
disp(results);

% Plot User Capacity Comparison
figure;
bar(supportedUsers);
set(gca, 'XTick', 1:3, 'XTickLabel', method);

xlabel('Multiple Access Method');
ylabel('Maximum Supported Users');
title('FDMA, TDMA and CDMA Capacity Comparison');
grid on;

% Plot CDMA SIR versus Number of Users
figure;
plot(K, sirDb, 'o-', 'LineWidth', 1.5);
hold on;

yline(targetSIRdB, '--r', 'Target SIR');

xlabel('Number of Simultaneous CDMA Users (K)');
ylabel('Estimated SIR (dB)');
title('CDMA SIR versus Number of Users');

grid on;
hold off;


%% PROGRAM 2: MOBILE-GENERATION COMPARISON

generation = {'1G'; '2G'; '2G'; '3G'; '4G'; '5G'};

system = {'AMPS'; 'GSM'; 'IS-95'; 'UMTS'; 'LTE'; '5G NR'};

accessMethod = {'FDMA'; 'TDMA'; 'CDMA'; ...
                'WCDMA'; 'OFDMA'; 'OFDMA-based'};

keyIdea = {'Separate frequency channels'; ...
           'Separate time slots'; ...
           'Code-based user separation'; ...
           'Wideband code-based access'; ...
           'Orthogonal frequency resources'; ...
           'Flexible OFDM-based radio access'};

% Create and Display Generation Table
evolutionTable = table(generation, system, accessMethod, keyIdea);

fprintf('\n============================================\n');
fprintf(' MOBILE GENERATION COMPARISON\n');
fprintf('============================================\n');

disp(evolutionTable);


%% PROGRAM 3: RESOURCE-ALLOCATION DIAGRAMS

figure;

% Use a 1-by-3 layout
tiledlayout(1, 3);

% Generate Colors for Users
colors = lines(4);


% FDMA: Different Users Use Different Frequency Channels
nexttile;
hold on;

for u = 1:4
    rectangle('Position', [0, u-1, 1, 0.8], ...
        'FaceColor', colors(u,:), ...
        'EdgeColor', 'k');

    text(0.5, u-0.6, sprintf('User %d', u), ...
        'HorizontalAlignment', 'center');
end

xlim([0 1]);
ylim([0 4]);

xlabel('Time');
ylabel('Frequency Channel');
title('FDMA');

set(gca, 'YTick', []);
hold off;


% TDMA: Different Users Use Different Time Slots
nexttile;
hold on;

for u = 1:4
    rectangle('Position', [u-1, 0, 0.8, 1], ...
        'FaceColor', colors(u,:), ...
        'EdgeColor', 'k');

    text(u-0.6, 0.5, sprintf('User %d', u), ...
        'HorizontalAlignment', 'center');
end

xlim([0 4]);
ylim([0 1]);

xlabel('Time Slot');
ylabel('Frequency Channel');
title('TDMA');

set(gca, 'YTick', []);
hold off;


% CDMA: Users Share the Same Time and Frequency Band
nexttile;
hold on;

for u = 1:4
    rectangle('Position', [0, 0, 1, 1], ...
        'FaceColor', colors(u,:), ...
        'FaceAlpha', 0.22, ...
        'EdgeColor', colors(u,:), ...
        'LineWidth', 1.5);
end

text(0.5, 0.62, 'Users 1-4 share', ...
    'HorizontalAlignment', 'center');

text(0.5, 0.40, 'band; different codes', ...
    'HorizontalAlignment', 'center');

xlim([0 1]);
ylim([0 1]);

xlabel('Time');
ylabel('Frequency Band');
title('CDMA');

set(gca, 'YTick', []);
hold off;

sgtitle('Conceptual Time-Frequency Resource Allocation');


%% PROGRAM 4: DISPLAY FINAL SUMMARY

fprintf('\n============================================\n');
fprintf(' EXPERIMENT COMPLETED SUCCESSFULLY\n');
fprintf('============================================\n');

fprintf('FDMA estimated capacity: %d users\n', fdmaUsers);
fprintf('TDMA estimated capacity: %d users\n', tdmaUsers);
fprintf('CDMA estimated capacity: %d users\n', cdmaUsers);

fprintf('\nGenerated outputs:\n');
fprintf('1. User capacity comparison graph\n');
fprintf('2. CDMA SIR versus number of users graph\n');
fprintf('3. Mobile-generation comparison table\n');
fprintf('4. FDMA, TDMA and CDMA resource-allocation diagrams\n');

fprintf('\nNote: Results use a simplified illustrative model.\n');
